import React, { useRef } from 'react';

export default function BackupManager({ appData, updateAppData }) {
    const fileInputRef = useRef(null);

    const exportData = () => {
        try {
            const dataStr = JSON.stringify(appData, null, 2);
            const blob = new Blob([dataStr], { type: 'application/json' });
            const url = URL.createObjectURL(blob);
            const a = document.createElement('a');
            a.href = url;
            a.download = `Zeiterfassung_Backup_${new Date().toISOString().split('T')[0]}.json`;
            document.body.appendChild(a);
            a.click();
            document.body.removeChild(a);
            URL.revokeObjectURL(url);
        } catch (error) {
            alert("Fehler beim Erstellen der Backup-Datei.");
        }
    };

    const handleFileImport = async (event) => {
        const file = event.target.files[0];
        if (!file) return;
        const reader = new FileReader();
        reader.onload = async (e) => {
            try {
                const parsed = JSON.parse(e.target.result);
                if (parsed && parsed.entries && parsed.settings) {
                    if (window.confirm("⚠️ Warnung: Dies überschreibt alle aktuellen Daten in der App mit dem Backup. Fortfahren?")) {
                        await updateAppData(parsed);
                        localStorage.setItem('zeiterfassung_db', JSON.stringify(parsed));
                        alert("✅ Backup erfolgreich wiederhergestellt!");
                        window.location.reload();
                    }
                } else {
                    alert("❌ Ungültige Backup-Datei. (Falsches Format)");
                }
            } catch(error) {
                alert("❌ Fehler beim Lesen der Datei. Sie scheint beschädigt zu sein.");
            }
            event.target.value = '';
        };
        reader.readAsText(file);
    };

    return (
        <div className="bg-indigo-50 p-4 rounded-xl shadow-sm space-y-3 border border-indigo-100">
            <h2 className="font-bold text-indigo-800 border-b border-indigo-200 pb-2 text-sm">Datensicherung (Backup)</h2>

                <input type="file" ref={fileInputRef} accept=".json" className="hidden" onChange={handleFileImport} />

export default function HolidaySettings({ appData, updateAppData }) {
    const [isFetchingHolidays, setIsFetchingHolidays] = useState(false);
    const bundesland = appData.settings.state;

    const updateState = async (val) => {
        await updateAppData(prev => ({ ...prev, settings: { ...prev.settings, state: val } }));
    };

    const fetchHolidays = async () => {
        setIsFetchingHolidays(true);
        const year = new Date().getFullYear();
        
        try {
            const fetchedHolidays = { ...appData.holidays };

            const loadAPI = async (y, isSchool) => {
                const typeUrl = isSchool ? "SchoolHolidays" : "PublicHolidays";
                const url = `https://openholidaysapi.org/${typeUrl}?countryIsoCode=DE&languageIsoCode=DE&validFrom=${y}-01-01&validTo=${y}-12-31&subdivisionCode=DE-${bundesland}`;
                const res = await fetch(url);
                const data = await res.json();
                
                data.forEach(item => {
                    const name = item.name[0].text;
                    const start = new Date(item.startDate);
                    const end = new Date(item.endDate);
                    for (let dt = new Date(start); dt <= end; dt.setDate(dt.getDate() + 1)) {
                        const dStr = dt.toISOString().split('T')[0];
                        if (!fetchedHolidays[dStr] || (!isSchool && fetchedHolidays[dStr].isSchool)) {
                            fetchedHolidays[dStr] = { name: name, isSchool: isSchool };
                        }
                    }
                });
            };

            await loadAPI(year, false); await loadAPI(year + 1, false);
            await loadAPI(year, true); await loadAPI(year + 1, true);

            await updateAppData(prev => ({ ...prev, holidays: fetchedHolidays }));

        { id: 1, name: "Früh", begin: "06:00", end: "14:30", pause: 30 },
        { id: 2, name: "Spät", begin: "14:00", end: "22:30", pause: 30 }
    ],
    settings: {
        state: "NW", agSaldo: 0.0,
        model: { hours: 40, days: [1, 2, 3, 4, 5], percentage: 100 },
        protectHolidays: true
    },
    holidays: {},
    activePauseStart: null
};

export function AppProvider({ children }) {
    const [appData, setAppData] = useState(null);
    const [isLoading, setIsLoading] = useState(true);

    useEffect(() => {
        const initApp = async () => {
            await requestPersistence();
            try {
                const savedData = await loadFromDB();
                if (savedData) {
                    setAppData({ ...defaultState, ...savedData });
                } else {
                    // Fallback-Migration für alte LocalStorage-Nutzer
                    const lsData = localStorage.getItem('zeiterfassung_db');
                    if (lsData) {
                        const parsed = JSON.parse(lsData);
                        setAppData({ ...defaultState, ...parsed });
                        await saveToDB({ ...defaultState, ...parsed });
                    } else {
                        setAppData(defaultState);
                        await saveToDB(defaultState);
                    }
                }
            } catch (error) {
                console.error("Datenbank Fehler:", error);
                setAppData(defaultState);
            } finally {
                setIsLoading(false);
            }
        };
        initApp();
    }, []);

    // Ersetzt das alte manuelle saveData() – React State & IndexedDB bleiben 100% synchron
    const updateAppData = async (newDataOrUpdater) => {
        setAppData((prev) => {
            const updated = typeof newDataOrUpdater === 'function' ? newDataOrUpdater(prev) : newDataOrUpdater;
            saveToDB(updated).catch(console.error);
            return updated;
        });
    };

    return (
        <AppContext.Provider value={{ appData, updateAppData, isLoading }}>
            {!isLoading ? children : <div className="flex h-screen items-center justify-center font-bold text-slate-500">System startet...</div>}
        </AppContext.Provider>
    );
}

/**
 * src/hooks/useTimeTracker.js
 * Custom Hook für die Interaktion mit dem täglichen Zeiterfassungs-State.
 * Nutzt den globalen AppContext und die isolierten Zeit-Utilities.
 */
import { useApp } from '../context/AppContext';
import { 
    getLocalDateString, 
    calculateDailySoll, 
    calculateNettoHours,
    suggestBreak,
    calculateMinutes
} from '../utils/timeCalculations';

export function useTimeTracker(selectedDateString) {
    const { appData, updateAppData } = useApp();

    // Lade den Eintrag für den ausgewählten Tag oder generiere einen leeren Dummy-Eintrag
    const entry = appData.entries[selectedDateString] || {
        begin: '',
        end: '',
        pause: '',
        soll: calculateDailySoll(selectedDateString, appData.settings, appData.holidays),
        remark: ''
    };

    // Helfer für Feiertage an diesem Tag
    const holiday = appData.holidays[selectedDateString];

    /**
     * Speichert oder aktualisiert den Eintrag für den ausgewählten Tag in der Datenbank.
     * Wenn alle Felder (außer Soll) leer sind und das Soll 0 ist, wird der Eintrag gelöscht, um Platz zu sparen.
     * 
     * @param {object} updatedData - Partielle Daten (z.B. { begin: "08:00", pause: 30 })
     */
    const saveEntry = async (updatedData) => {
        const mergedEntry = { ...entry, ...updatedData };
        
        // Berechne Netto und checke, ob wir eine leere Hülle löschen können
        const isCompletelyEmpty = !mergedEntry.begin && !mergedEntry.end && !mergedEntry.remark && mergedEntry.soll === 0;

        await updateAppData((prev) => {
            const newEntries = { ...prev.entries };
            
            if (isCompletelyEmpty) {
                delete newEntries[selectedDateString];
            } else {
                newEntries[selectedDateString] = mergedEntry;
            }
            
            return { ...prev, entries: newEntries };
        });
    };

    /**
     * Setzt den aktuellen Eintrag komplett zurück (inklusive Soll-Neuberechnung).
     */
    const clearEntry = async () => {
        const defaultSoll = calculateDailySoll(selectedDateString, appData.settings, appData.holidays);
        
        if (defaultSoll === 0) {
            // Wenn Soll 0 ist, können wir den Eintrag direkt aus der DB putzen
             await updateAppData((prev) => {
                const newEntries = { ...prev.entries };
                delete newEntries[selectedDateString];
                return { ...prev, entries: newEntries };
            });
        } else {
            // Andernfalls setzen wir ihn auf Standard-Werte zurück
            await saveEntry({
                begin: '',
                end: '',
                pause: '',
                soll: defaultSoll,
                remark: ''
            });
        }
    };

    /**
     * Wendet eine Vorlage (Template) auf den aktuellen Tag an.
     * @param {object} template - Z.B. { begin: "06:00", end: "14:30", pause: 30 }
     */
    const applyTemplate = async (template) => {
        await saveEntry({
            begin: template.begin,
            end: template.end,
            pause: template.pause || 30
        });
    };

    /**
     * "Kommen/Gehen" Toggle-Logik (Stempeln).
     * Nimmt die aktuelle Uhrzeit und trägt sie in "begin" ein, falls leer.
     * Andernfalls in "end". Berechnet optional automatisch eine Pausenempfehlung.
     */
    const handleQuickStamp = async () => {
        const now = new Date();
        const hh = String(now.getHours()).padStart(2, '0');
        const mm = String(now.getMinutes()).padStart(2, '0');
        const timeStr = `${hh}:${mm}`;

        let newBegin = entry.begin;
        let newEnd = entry.end;
        let newPause = parseFloat(entry.pause) || 0;

        if (!entry.begin) {
            newBegin = timeStr;
        } else {
            newEnd = timeStr;
            // Automatische Pause vorschlagen, falls noch keine gesetzt wurde
            if (!entry.pause) {
                const diffMin = calculateMinutes(newBegin, newEnd);
                newPause = suggestBreak(diffMin);
            }
        }

        await saveEntry({ begin: newBegin, end: newEnd, pause: newPause });
    };

    // Live-Berechnungen (abgeleiteter State) für die UI (View)
    const nettoHours = calculateNettoHours(entry.begin, entry.end, parseFloat(entry.pause) || 0);
    const currentSoll = parseFloat(entry.soll) || 0;
    const diffHours = nettoHours - currentSoll;

    return {
        entry,          // Die Daten des aktuellen Tages (begin, end, pause, soll, remark)
        holiday,        // Gibt es einen Feiertag / Schulferien an diesem Tag?
        nettoHours,     // Berechnete Nettozeit in Stunden (float)
        diffHours,      // Differenz zum Soll in Stunden (float, negativ bei Minusstunden)
        saveEntry,      // Funktion zum Speichern (Teil-Updates erlaubt)
        clearEntry,     // Setzt den Tag zurück
        applyTemplate,  // Wendet Schicht-Muster an
        handleQuickStamp// Kommen/Gehen Button-Logik
    };
}

const DB_NAME = 'ZeiterfassungDB';
const STORE_NAME = 'appState';
const DB_VERSION = 1;

export async function requestPersistence() {
    if (navigator.storage && navigator.storage.persist) {
        const isPersisted = await navigator.storage.persist();
        console.info(`Storage Persistence: ${isPersisted ? 'Aktiv' : 'Nicht garantiert'}`);
    }
}

export function getDB() {
    return new Promise((resolve, reject) => {
        const request = indexedDB.open(DB_NAME, DB_VERSION);
        request.onupgradeneeded = (e) => {
            const db = e.target.result;
            if (!db.objectStoreNames.contains(STORE_NAME)) {
                db.createObjectStore(STORE_NAME);
            }
        };
        request.onsuccess = (e) => resolve(e.target.result);
        request.onerror = (e) => reject(e.error);
    });
}

export async function saveToDB(data) {
    const db = await getDB();
    return new Promise((resolve, reject) => {
        const tx = db.transaction(STORE_NAME, 'readwrite');
        tx.objectStore(STORE_NAME).put(data, 'main');
        tx.oncomplete = () => resolve();
        tx.onerror = (e) => reject(e.error);
    });
}

export async function loadFromDB() {
    const db = await getDB();
    return new Promise((resolve, reject) => {
        const tx = db.transaction(STORE_NAME, 'readonly');
        const request = tx.objectStore(STORE_NAME).get('main');
        request.onsuccess = () => resolve(request.result);
        request.onerror = (e) => reject(e.error);
    });
}

/**
 * src/utils/timeCalculations.js
 * Reine Business-Logik für die Zeitberechnung (Local-First, keine React/UI-Abhängigkeiten)
 */

export function calculateMinutes(begin, end) {
    if (!begin || !end) return 0;
    const [h1, m1] = begin.split(':').map(str => parseInt(str, 10) || 0);
    const [h2, m2] = end.split(':').map(str => parseInt(str, 10) || 0);
    let diff = (h2 * 60 + m2) - (h1 * 60 + m1);
    if (diff < 0) diff += 24 * 60; 
    return diff;
}

export function calculateNettoHours(begin, end, pauseMin = 0) {
    if (!begin || !end) return 0.0;
    const bruttoMin = calculateMinutes(begin, end);
    const nettoMin = bruttoMin - pauseMin;
    return Math.max(0, nettoMin / 60);
}

export function isWeekend(date) {
    const d = typeof date === 'string' ? new Date(date) : date;
    const day = d.getDay();
    return day === 0 || day === 6;
}

export function getLocalDateString(d) {
    if (!d) return "";
    const year = d.getFullYear();
    const month = String(d.getMonth() + 1).padStart(2, '0');
    const day = String(d.getDate()).padStart(2, '0');
    return `${year}-${month}-${day}`;
}

export function calculateDailySoll(dateString, settings, holidays) {
    if (!settings || !settings.model || !settings.model.days) return 0.0;
    const date = new Date(dateString);
    const dayOfWeek = date.getDay();
    const isWorkDay = settings.model.days.includes(dayOfWeek);
    
    if (!isWorkDay) return 0.0;
    
    if (settings.protectHolidays && holidays && holidays[dateString] && !holidays[dateString].isSchool) {
        return 0.0;
    }
    const percent = settings.model.percentage || 100;
    const effWeekly = settings.model.hours * (percent / 100);
    const hoursPerDay = effWeekly / Math.max(1, settings.model.days.length);
    return parseFloat(hoursPerDay.toFixed(2));
}

export function suggestBreak(diffMin) {
    if (diffMin > 9 * 60) return 45;
    if (diffMin > 6 * 60) return 30;
    return 0;
}

/**
 * Berechnet den ewigen Gesamtsaldo (App-Gesamtsaldo) vom ersten Eintrag bis zum gewählten Monat.
 */
export function calculateAllTimeAppDiff(entries, settings, holidays, viewYear, viewMonth) {
    const entryKeys = Object.keys(entries);
    if (entryKeys.length === 0) return 0;
    
    entryKeys.sort();
    const firstDateStr = entryKeys[0];
    const firstDate = new Date(firstDateStr);
    firstDate.setHours(0,0,0,0);
    
    const lastDayOfViewMonth = new Date(viewYear, viewMonth + 1, 0);
    const today = new Date();
    today.setHours(0,0,0,0);
    
    const endDate = lastDayOfViewMonth;
    let totalDiff = 0;
    let d = new Date(firstDate);
    
    let iterations = 0; 
    // Max 3000 Iterationen als Endlosschleifen-Schutz (reicht für >8 Jahre Historie)
    while (d <= endDate && iterations < 3000) {
        const dStr = getLocalDateString(d);
        const entry = entries[dStr];
        const daySoll = entry && entry.soll !== undefined ? entry.soll : calculateDailySoll(dStr, settings, holidays);
        
        let netStd = 0;
        if (entry && entry.begin && entry.end) {
            const bMin = calculateMinutes(entry.begin, entry.end);
            netStd = Math.max(0, (bMin - (parseFloat(entry.pause) || 0)) / 60);
        }
        
        let dayDiff = netStd - daySoll;
        const isFuture = d > today;
        
        if (!entry && isFuture) {
            dayDiff = 0;
        } else if (!entry && !isFuture && daySoll > 0) {
            dayDiff = -daySoll;
        }
        
        totalDiff += dayDiff;
        d.setDate(d.getDate() + 1);
        iterations++;
    }
    return totalDiff;
}

    const applyPattern = async () => {
        const daysInMonth = new Date(year, month + 1, 0).getDate();
        let count = 0;

        await updateAppData(prev => {
            const newEntries = { ...prev.entries };
            for (let i = 1; i <= daysInMonth; i++) {
                const dStr = `${year}-${String(month + 1).padStart(2, '0')}-${String(i).padStart(2, '0')}`;
                const tplId = pattern[new Date(dStr).getDay()];

                if (prev.settings.protectHolidays && prev.holidays[dStr] && !prev.holidays[dStr].isSchool) continue;

                if (tplId) {
                    const existing = newEntries[dStr];
                    if (!existing || overwritePattern) {
                        const t = prev.templates.find(x => x.id === Number(tplId));
                        if (t) {
                            newEntries[dStr] = {
                                begin: t.begin, end: t.end, pause: t.pause !== undefined ? t.pause : 30,
                                soll: existing && existing.soll !== undefined ? existing.soll : calculateDailySoll(dStr, prev.settings, prev.holidays),
                                remark: existing ? existing.remark : ''
                            };
                            count++;
                        }
                    }
                }
            }
            return { ...prev, entries: newEntries };
        });
        alert(`${count} Schichten in diesen Monat eingetragen!`);
    };

    const toggleProtectHolidays = async () => {
        await updateAppData(prev => ({
            ...prev, settings: { ...prev.settings, protectHolidays: !prev.settings.protectHolidays }
        }));
    };

    const cells = useMemo(() => {
        let firstDay = new Date(year, month, 1).getDay();
        firstDay = firstDay === 0 ? 7 : firstDay;
        const daysInMonth = new Date(year, month + 1, 0).getDate();
        const gridCells = [];

        for (let i = 1; i < firstDay; i++) gridCells.push({ type: 'empty', id: `empty-${i}` });

        for (let i = 1; i <= daysInMonth; i++) {
            const dStr = `${year}-${String(month + 1).padStart(2, '0')}-${String(i).padStart(2, '0')}`;
            const isWeekend = (new Date(dStr).getDay() === 0 || new Date(dStr).getDay() === 6);
            const entry = appData.entries[dStr];
            let tplMatch = null, netStd = 0;

            if (entry && entry.begin && entry.end) {
                const bMin = calculateMinutes(entry.begin, entry.end);
                netStd = Math.max(0, (bMin - (parseFloat(entry.pause) || 0)) / 60);
                tplMatch = appData.templates.find(t => t.begin === entry.begin && t.end === entry.end && (t.pause || 0) === (entry.pause || 0));
            }
            gridCells.push({ type: 'day', id: dStr, dayNum: i, isWeekend, entry, tplMatch, netStd });
        }
        
        while (gridCells.length < 42) gridCells.push({ type: 'empty', id: `empty-end-${gridCells.length}` });
        return gridCells;
    }, [year, month, appData.entries, appData.templates]);

    return (
        <div className="flex flex-col h-full gap-2">
            <PaintPalette 
                templates={appData.templates} activePaintTemplate={activePaintTemplate} 
                setActivePaintTemplate={setActivePaintTemplate} colors={plannerColors} 
            />
            <PlanerGrid 
                cells={cells} monthName={monthName} changeMonth={changeMonth} 
                handleCellClick={handleCellClick} activePaintTemplate={activePaintTemplate} 
                templates={appData.templates} colors={plannerColors} 
            />
            <AutoFillMenu 
                templates={appData.templates} protectHolidays={appData.settings.protectHolidays} 
                toggleProtectHolidays={toggleProtectHolidays} pattern={pattern} setPattern={setPattern} 
                overwritePattern={overwritePattern} setOverwritePattern={setOverwritePattern} applyPattern={applyPattern} 
            />
        </div>
    );
}

import React, { useState, useMemo } from 'react';
import { useApp } from '../../context/AppContext';
import { calculateMinutes } from '../../utils/timeCalculations';
import { exportMonthPDF } from '../../services/pdfExport';

export default function StatistikView() {
    const { appData } = useApp();
    const [viewDate, setViewDate] = useState(new Date());
    const [statMode, setStatMode] = useState('month'); // 'month', 'quarter', 'year'

    const changeStatPeriod = (dir) => {
        const newDate = new Date(viewDate);
        if (statMode === 'month') {
            newDate.setMonth(newDate.getMonth() + dir);
        } else if (statMode === 'quarter') {
            newDate.setMonth(newDate.getMonth() + (dir * 3));
        } else if (statMode === 'year') {
            newDate.setFullYear(newDate.getFullYear() + dir);
        }
        setViewDate(newDate);
    };

    const year = viewDate.getFullYear();
    const month = viewDate.getMonth();

    // Dynamische Perioden-Titel
    let startMonth, endMonth, titleStr;
    if (statMode === 'month') {
        startMonth = month;
        endMonth = month;
        titleStr = viewDate.toLocaleString('de-DE', { month: 'long', year: 'numeric' });
    } else if (statMode === 'quarter') {
        const q = Math.floor(month / 3);
        startMonth = q * 3;
        endMonth = startMonth + 2;
        titleStr = `Q${q + 1} ${year}`;
    } else if (statMode === 'year') {
        startMonth = 0;
        endMonth = 11;
        titleStr = `Jahr ${year}`;
    }

    // Statistik-Berechnungen durch useMemo gecached
    const stats = useMemo(() => {
        let totalNet = 0;
        let daysWorked = 0;
        let totalDaysInPeriod = 0;
        let tplDistribution = {};

        appData.templates.forEach(t => tplDistribution[t.name] = 0);
        tplDistribution['Individuell / Sonstige'] = 0;

        for (let m = startMonth; m <= endMonth; m++) {
            const daysInMonth = new Date(year, m + 1, 0).getDate();
            totalDaysInPeriod += daysInMonth;

            for (let i = 1; i <= daysInMonth; i++) {
                const dStr = `${year}-${String(m+1).padStart(2,'0')}-${String(i).padStart(2,'0')}`;
                const entry = appData.entries[dStr];

                if (entry && entry.begin && entry.end) {
                    const bMin = calculateMinutes(entry.begin, entry.end);
                    const netStd = Math.max(0, (bMin - (parseFloat(entry.pause) || 0)) / 60);
                    if (netStd > 0) {
                        totalNet += netStd;
                        daysWorked++;

                        const matchedTpl = appData.templates.find(t => t.begin === entry.begin && t.end === entry.end && (parseFloat(t.pause) || 0) === (parseFloat(entry.pause) || 0));
                        if (matchedTpl) {
                            tplDistribution[matchedTpl.name] += netStd;
                        } else {
                            tplDistribution['Individuell / Sonstige'] += netStd;
                        }
                    }
                }
            }
        }

        const avgDay = daysWorked > 0 ? (totalNet / daysWorked) : 0;
        const avgWeek = totalNet / (totalDaysInPeriod / 7);

        const percent = appData.settings.model.percentage || 100;
        const effWeekly = appData.settings.model.hours * (percent / 100);
        const hrMonthlySoll = effWeekly * 13 / 3;
        
        let hrPeriodSoll = 0;
        if (statMode === 'month') hrPeriodSoll = hrMonthlySoll;
        else if (statMode === 'quarter') hrPeriodSoll = hrMonthlySoll * 3;
        else if (statMode === 'year') hrPeriodSoll = hrMonthlySoll * 12;

        const sortedTpls = Object.keys(tplDistribution)
            .map(k => ({name: k, hours: tplDistribution[k]}))
            .filter(x => x.hours > 0)
            .sort((a,b) => b.hours - a.hours);

        return { totalNet, hrPeriodSoll, avgDay, avgWeek, sortedTpls };
    }, [year, startMonth, endMonth, appData.entries, appData.templates, appData.settings, statMode]);

    const handleExportPDF = () => {
        const monthName = viewDate.toLocaleString('de-DE', { month: 'long' });
        exportMonthPDF(appData, year, month, monthName);
    };

    return (
        <div className="flex flex-col h-full gap-3">
            <div className="bg-blue-600 p-2 rounded-xl shadow-sm shrink-0 flex flex-col gap-2">
                <div className="flex justify-between items-center text-white">
                    <button onClick={() => changeStatPeriod(-1)} className="p-2 bg-blue-500 rounded-lg active:bg-blue-700">
                        <svg className="w-5 h-5" fill="none" stroke="currentColor" viewBox="0 0 24 24"><path strokeLinecap="round" strokeLinejoin="round" strokeWidth="2" d="M15 19l-7-7 7-7"></path></svg>
                    </button>
                    <div className="text-base font-bold">{titleStr}</div>
                    <button onClick={() => changeStatPeriod(1)} className="p-2 bg-blue-500 rounded-lg active:bg-blue-700">
                        <svg className="w-5 h-5" fill="none" stroke="currentColor" viewBox="0 0 24 24"><path strokeLinecap="round" strokeLinejoin="round" strokeWidth="2" d="M9 5l7 7-7 7"></path></svg>
                    </button>
                </div>
                
                <div className="flex bg-blue-700/50 p-1 rounded-lg">
                    {['month', 'quarter', 'year'].map(mode => (
                        <button 
                            key={mode}
                            onClick={() => setStatMode(mode)} 
                            className={`flex-1 text-[10px] font-bold py-1.5 rounded-md transition-all ${statMode === mode ? 'bg-white text-blue-600 shadow-sm' : 'text-blue-100 hover:text-white'}`}
                        >
                            {mode === 'month' ? 'Monat' : mode === 'quarter' ? 'Quartal' : 'Jahr'}
                        </button>
                    ))}
                </div>
            </div>

            <div className="grid grid-cols-2 gap-2 shrink-0">
                {[
                    { label: 'Gesamt (Ist)', value: stats.totalNet, color: 'text-blue-600' },
                    { label: 'HR Soll (Formel)', value: stats.hrPeriodSoll, color: 'text-slate-700' },
                    { label: 'Ø pro Tag', value: stats.avgDay, color: 'text-indigo-600' },
                    { label: 'Ø pro Woche', value: stats.avgWeek, color: 'text-emerald-600' },
                ].map((kpi, idx) => (
                    <div key={idx} className="bg-white py-3 px-1 rounded-xl shadow-sm border border-slate-100 flex flex-col items-center justify-center">
                        <div className="text-[9px] font-bold text-slate-400 uppercase tracking-widest mb-0.5">{kpi.label}</div>
                        <div className={`text-lg font-black ${kpi.color}`}>{kpi.value.toFixed(1)} h</div>
                    </div>
                ))}
            </div>

            <div className="bg-white p-3 rounded-xl shadow-sm border border-slate-100 flex-1 flex flex-col min-h-0">
                <h2 className="font-bold text-slate-800 text-xs mb-3 shrink-0 border-b border-slate-100 pb-2 uppercase tracking-wide">
                    Verteilung der Arbeitszeit
                </h2>
                <div className="flex-1 overflow-y-auto space-y-4 pr-1 no-scrollbar">
                    {stats.sortedTpls.length === 0 ? (
                        <div className="text-xs font-medium text-slate-400 text-center py-6">Keine Arbeitszeiten in diesem Zeitraum erfasst.</div>
                    ) : (
                        stats.sortedTpls.map(item => {
                            const percent = (item.hours / stats.totalNet) * 100;
                            return (
                                <div key={item.name} className="mb-3">
                                    <div className="flex justify-between text-[11px] font-bold mb-1.5 px-1">
                                        <span className="text-slate-700">{item.name}</span>
                                        <span className="text-blue-600">{item.hours.toFixed(1)} h <span className="text-slate-400 font-normal ml-1">({percent.toFixed(0)}%)</span></span>
                                    </div>
                                    <div className="w-full bg-slate-100 rounded-full h-2 overflow-hidden border border-slate-200">
                                        <div className="bg-blue-500 h-2 rounded-full" style={{ width: `${percent}%` }}></div>
                                    </div>
                                </div>
                            );
                        })
                    )}
                </div>
            </div>

            <div className="shrink-0 mt-auto">
                <button onClick={handleExportPDF} className="w-full py-3.5 bg-indigo-600 text-white font-bold rounded-xl shadow-md flex items-center justify-center gap-2 active:bg-indigo-700 text-sm transition-colors">
                    <svg className="w-5 h-5" fill="none" stroke="currentColor" viewBox="0 0 24 24"><path strokeLinecap="round" strokeLinejoin="round" strokeWidth="2" d="M12 10v6m0 0l-3-3m3 3l3-3m2 8H7a2 2 0 01-2-2V5a2 2 0 012-2h5.586a1 1 0 01.707.293l5.414 5.414a1 1 0 01.293.707V19a2 2 0 01-2 2z"></path></svg>
                    Gewählten Monat als PDF exportieren
                </button>
            </div>
        </div>
    );
}

import { useState } from 'react';
import { useApp } from './context/AppContext.jsx';
import { getLocalDateString } from './utils/timeCalculations.js';
import BottomNav from './components/layout/BottomNav.jsx';

import HeuteView from './components/views/HeuteView.jsx';
import KalenderView from './components/views/KalenderView.jsx';
import PlanerView from './components/views/PlanerView.jsx';
import StatistikView from './components/views/StatistikView.jsx';

export default function App() {
    const { appData } = useApp();
    const [activeTab, setActiveTab] = useState('heute');
    
    // Globaler Datums-State, der von Kalender zu Heute gereicht wird
    const [selectedDate, setSelectedDate] = useState(getLocalDateString(new Date()));

            <main className="flex-1 overflow-y-auto w-full max-w-lg mx-auto relative pt-3 pb-2 px-3 flex flex-col">

            <div className="w-full max-w-lg mx-auto shrink-0 z-20">
